\makeatletter
\def\myparseISO#1-#2-#3\relax{%
  \def\anno{#1}%
  \def\mese{#2}%
  \def\giorno{#3}%
}

\newcommand{\formatISO}[1]{%
  % Prima rimuoviamo spazi e newline
  \edef\tempdate{#1}%
  \ifthenelse{\equal{\tempdate}{}}{%
    % Se vuoto, non fare nulla
  }{%
    \expandafter\myparseISO\tempdate\relax
    \def\myMese{}%
    \ifthenelse{\equal{\mese}{01}}{\def\myMese{gen.}}{}%
    \ifthenelse{\equal{\mese}{02}}{\def\myMese{feb.}}{}%
    \ifthenelse{\equal{\mese}{03}}{\def\myMese{mar.}}{}%
    \ifthenelse{\equal{\mese}{04}}{\def\myMese{apr.}}{}%
    \ifthenelse{\equal{\mese}{05}}{\def\myMese{mag.}}{}%
    \ifthenelse{\equal{\mese}{06}}{\def\myMese{giu.}}{}%
    \ifthenelse{\equal{\mese}{07}}{\def\myMese{lug.}}{}%
    \ifthenelse{\equal{\mese}{08}}{\def\myMese{ago.}}{}%
    \ifthenelse{\equal{\mese}{09}}{\def\myMese{set.}}{}%
    \ifthenelse{\equal{\mese}{10}}{\def\myMese{ott.}}{}%
    \ifthenelse{\equal{\mese}{11}}{\def\myMese{nov.}}{}%
    \ifthenelse{\equal{\mese}{12}}{\def\myMese{dic.}}{}%
    \giorno\space\myMese\space\anno
  }%
}

% Comando wrapper che gestisce meglio i placeholder
\newcommand{\safeFormatISO}[1]{%
  \def\temparg{#1}%
  % Rimuove spazi e controlla se vuoto
  \expandafter\ifx\expandafter\relax\detokenize{#1}\relax
    % Vuoto, non fare nulla
  \else
    \formatISO{#1}%
  \fi
}
\makeatother
